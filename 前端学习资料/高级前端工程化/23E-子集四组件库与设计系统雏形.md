## 步骤二：Button

`packages/ui/src/button/Button.tsx`：

```typescript
import type { ButtonHTMLAttributes, ReactNode } from 'react'

export type ButtonVariant = 'primary' | 'secondary' | 'danger'

export type ButtonProps = {
  variant?: ButtonVariant
  children: ReactNode
} & ButtonHTMLAttributes<HTMLButtonElement>

const variantClassName: Record<ButtonVariant, string> = {
  primary: 'acme-button--primary',
  secondary: 'acme-button--secondary',
  danger: 'acme-button--danger'
}

export function Button({ variant = 'primary', className = '', children, ...rest }: ButtonProps) {
  return (
    <button className={['acme-button', variantClassName[variant], className].join(' ')} {...rest}>
      {children}
    </button>
  )
}
```

`packages/ui/src/button/index.ts`：

```typescript

```

## 步骤三：DataTable

`packages/ui/src/data-table/DataTable.tsx`：

```typescript
import type { ReactNode } from 'react'

export type DataTableColumn<TRow> = {
  key: string
  title: string
  render: (row: TRow) => ReactNode
}

export type DataTableProps<TRow> = {
  rows: TRow[]
  columns: Array<DataTableColumn<TRow>>
  getRowKey?: (row: TRow, index: number) => string
  emptyText?: string
}

export function DataTable<TRow>({
  rows,
  columns,
  getRowKey = (_, index) => String(index),
  emptyText = 'No data'
}: DataTableProps<TRow>) {
  if (rows.length === 0) {
    return <div role="status">{emptyText}</div>
  }

  return (
    <table>
      <thead>
        <tr>
          {columns.map((column) => (
            <th key={column.key}>{column.title}</th>
          ))}
        </tr>
      </thead>
      <tbody>
        {rows.map((row, rowIndex) => (
          <tr key={getRowKey(row, rowIndex)}>
            {columns.map((column) => (
              <td key={column.key}>{column.render(row)}</td>
            ))}
          </tr>
        ))}
      </tbody>
    </table>
  )
}
```

`packages/ui/src/data-table/index.ts`：


## 步骤四：StatusBadge

```typescript
export type StatusBadgeStatus = 'active' | 'disabled' | 'open' | 'closed'

export type StatusBadgeProps = {
  status: StatusBadgeStatus
}

const labelMap: Record<StatusBadgeStatus, string> = {
  active: 'Active',
  disabled: 'Disabled',
  open: 'Open',
  closed: 'Closed'
}

export function StatusBadge({ status }: StatusBadgeProps) {
  return <span className={`acme-status acme-status--${status}`}>{labelMap[status]}</span>
}
```

## 验收标准

- `admin` 和 `portal` 都能从 `@acme/ui` 引入组件。
- `@acme/ui` 不依赖 `apps/*`。
- `react` 和 `react-dom` 是 peer dependencies。
- `dist/index.d.ts` 能生成。
- `DataTable` 泛型能推断 row 类型。
